def next_delivery_date(scanned_at: datetime, working_days: int = 2) -> date:
    """Return the promised delivery date, skipping weekends."""
    day = scanned_at.date()
    added = 0
    while added < working_days:
        day += timedelta(days=1)
        if day.weekday() < 5:
            added += 1
    return day
